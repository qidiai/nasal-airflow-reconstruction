{"schema":"consent_record/v1","timestamp":"ISO8601","prompt_version":"T17E-v1",
"case_id":"<case>","data_subject":"self|authorized_third_party|demo",
"source_declaration":"<来源一句话，demo 可空>","authorization_ref":"<他人数据必填，self/demo 空>",
"processing_scope":"local_only|cloud_optin","retention":"session_only|days:N",
